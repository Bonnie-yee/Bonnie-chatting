import fs from 'node:fs/promises';
import path from 'node:path';
import http from 'node:http';
import { randomBytes, randomUUID, createHash, timingSafeEqual } from 'node:crypto';
import { createRemoteJWKSet, jwtVerify } from 'jose';

const ISSUER = 'https://auth.openai.com';
const RESOURCE = 'https://api.openai.com/v1';
const SCOPES = 'openid profile email offline_access resource.invoke chatgpt.tokens.use.direct';
const jwks = createRemoteJWKSet(new URL(`${ISSUER}/.well-known/jwks.json`));

export async function privateJson(file, value) {
  await fs.mkdir(path.dirname(file), { recursive: true, mode: 0o700 });
  const temporary = `${file}.${randomUUID()}.tmp`;
  await fs.writeFile(temporary, JSON.stringify(value, null, 2) + '\n', { mode: 0o600 });
  await fs.rename(temporary, file);
  await fs.chmod(file, 0o600);
}

async function optionalJson(file) {
  try { return JSON.parse(await fs.readFile(file, 'utf8')); }
  catch (error) { if (error.code === 'ENOENT') return null; throw error; }
}

export async function validateIdentity(token, clientId, nonce, keySet = jwks) {
  const { payload } = await jwtVerify(token, keySet, { issuer: ISSUER, audience: clientId, algorithms: ['RS256'], requiredClaims: ['iss', 'sub', 'aud', 'exp', 'nonce'], clockTolerance: 5 });
  if (!payload.sub || payload.nonce !== nonce) throw new Error('登录身份核对失败，请重新登录。');
  if (Array.isArray(payload.aud) && payload.aud.length > 1 && payload.azp !== clientId) throw new Error('登录授权对象不匹配。');
  return { subject: payload.sub, issuer: payload.iss, email: payload.email ?? null };
}

export function sameState(actual, expected) {
  if (typeof actual !== 'string' || typeof expected !== 'string') return false;
  const a = Buffer.from(actual), b = Buffer.from(expected);
  return a.length === b.length && timingSafeEqual(a, b);
}

export class ChatGPTAccount {
  constructor(directory, { fetchImpl = fetch, verifyIdentity = validateIdentity, expectedEmail = null } = {}) {
    // This app never loads Codex/CLI credentials or environment API keys.
    this.directory = directory;
    this.fetch = fetchImpl;
    this.verifyIdentity = verifyIdentity;
    this.expectedEmail = expectedEmail?.trim().toLowerCase() ?? null;
    this.pending = null;
    this.refreshing = null;
    this.lastError = null;
  }

  async init() {
    await fs.mkdir(this.directory, { recursive: true, mode: 0o700 });
    await fs.chmod(this.directory, 0o700);
    const hostPath = path.join(this.directory, 'host.json');
    this.host = await optionalJson(hostPath);
    if (!this.host) {
      this.host = { ext_agent_host_id: `urn:uuid:${randomUUID()}` };
      await privateJson(hostPath, this.host);
    }
    this.active = await optionalJson(path.join(this.directory, 'active.json'));
    this.records = await optionalJson(path.join(this.directory, 'registrations.json')) ?? [];
    this.credentials = this.active ? await optionalJson(this.credentialPath(this.active.id)) : null;
  }

  credentialPath(id) {
    if (!/^[a-f0-9]{64}$/.test(id)) throw new Error('Invalid local registration ID');
    return path.join(this.directory, `connection-${id}.json`);
  }

  view() {
    const record = this.records.find(row => row.id === this.active?.id);
    return {
      connected: Boolean(this.credentials?.access_token),
      sharing: Boolean(this.credentials?.access_token && this.credentials.scopes?.includes('chatgpt.tokens.use.direct')),
      confirmed: Boolean(this.credentials?.access_token && this.active?.confirmed),
      email: record?.email ?? null,
      registrations: this.records.map(({ id, email }) => ({ id, email })),
      activeId: this.active?.id ?? null,
      pending: Boolean(this.pending),
      error: this.lastError
    };
  }

  async select(id) {
    if (this.pending) throw new Error('请先完成正在进行的登录。');
    if (!this.records.some(row => row.id === id)) throw new Error('找不到这个助手账号。');
    this.credentials = await optionalJson(this.credentialPath(id));
    this.active = { id, confirmed: false };
    await privateJson(path.join(this.directory, 'active.json'), this.active);
  }

  async confirmOwner() {
    if (!this.credentials || this.pending) throw new Error('请先完成自己账号的登录。');
    if (this.expectedEmail && this.credentials.email?.toLowerCase() !== this.expectedEmail) throw new Error('当前登录的邮箱不是你指定的账号。');
    this.active = { id: this.credentials.id, confirmed: true };
    await privateJson(path.join(this.directory, 'active.json'), this.active);
  }

  async startLogin({ newAccount = false } = {}) {
    if (this.pending) return { pending: true };
    this.lastError = null;
    const selected = newAccount ? null : this.records.find(row => row.id === this.active?.id);
    const state = randomBytes(32).toString('base64url');
    const nonce = randomBytes(32).toString('base64url');
    const verifier = randomBytes(48).toString('base64url');
    const listener = http.createServer((req, res) => this.callback(req, res).catch(() => {
      this.lastError ||= '登录未完成，请重新尝试。';
      if (!res.headersSent) res.writeHead(400, { 'Content-Type': 'text/plain; charset=utf-8' });
      res.end(this.lastError);
      this.finishPending();
    }));
    await new Promise((resolve, reject) => { listener.once('error', reject); listener.listen(0, '127.0.0.1', resolve); });
    const redirect = `http://127.0.0.1:${listener.address().port}/auth/callback`;
    const timer = setTimeout(() => { this.lastError = '登录等待已结束，请重新登录。'; this.finishPending(); }, 5 * 60 * 1000);
    timer.unref();
    this.pending = { state, nonce, verifier, listener, redirect, selected, timer };
    const url = new URL(`${ISSUER}/api/accounts/authorize`);
    const fields = {
      client_id: selected?.client_id ?? 'dynamic_agent_client',
      ext_agent_host_id: this.host.ext_agent_host_id,
      response_type: 'code', redirect_uri: redirect, scope: SCOPES, resource: RESOURCE,
      state, nonce, code_challenge_method: 'S256',
      code_challenge: createHash('sha256').update(verifier).digest('base64url')
    };
    if (!selected) fields.agent_name_hint = '暧昧回复助手';
    // No ID-token hint: the user must see and choose the account, especially on a shared Mac.
    for (const [key, value] of Object.entries(fields)) url.searchParams.set(key, value);
    return { authUrl: url.href, pending: true };
  }

  finishPending() {
    if (this.pending) { clearTimeout(this.pending.timer); this.pending.listener.close(); this.pending = null; }
  }

  async callback(req, res) {
    const url = new URL(req.url, 'http://127.0.0.1');
    if (req.method !== 'GET' || url.pathname !== '/auth/callback') { res.writeHead(404); res.end(); return; }
    const pending = this.pending;
    if (!pending || pending.consumed || !sameState(url.searchParams.get('state'), pending.state)) {
      res.writeHead(400, { 'Content-Type': 'text/plain; charset=utf-8' }); res.end('这次登录请求无效，请从助手重新登录。'); return;
    }
    pending.consumed = true;
    if (url.searchParams.has('error')) throw new Error('Authorization denied');
    const clientId = url.searchParams.get('client_id') ?? pending.selected?.client_id;
    if (!clientId || clientId === 'dynamic_agent_client' || clientId.length > 256 || !url.searchParams.get('code')) throw new Error('Incomplete registration');
    if (pending.selected && clientId !== pending.selected.client_id) throw new Error('Registration mismatch');
    const tokens = await this.tokenRequest({
      grant_type: 'authorization_code', client_id: clientId, code: url.searchParams.get('code'),
      code_verifier: pending.verifier, redirect_uri: pending.redirect, resource: RESOURCE
    });
    if (!tokens.id_token || !tokens.access_token || !Number.isFinite(tokens.expires_in) || tokens.expires_in <= 0) throw new Error('Incomplete token response');
    const identity = await this.verifyIdentity(tokens.id_token, clientId, pending.nonce);
    if (this.expectedEmail && identity.email?.toLowerCase() !== this.expectedEmail) {
      this.lastError = '登录回来的邮箱不是你指定的账号。请重新登录自己的账号。';
      throw new Error('Wrong account');
    }
    if (pending.selected && (identity.subject !== pending.selected.subject || identity.issuer !== pending.selected.issuer)) throw new Error('Account identity changed');
    const id = createHash('sha256').update(`${identity.issuer}\0${identity.subject}\0${clientId}`).digest('hex');
    const record = { id, client_id: clientId, ...identity };
    const credentials = {
      ...record, ext_agent_host_id: this.host.ext_agent_host_id,
      access_token: tokens.access_token, refresh_token: tokens.refresh_token ?? null, id_token: tokens.id_token,
      scopes: String(tokens.scope ?? '').split(' ').filter(Boolean), expires_at: Date.now() + tokens.expires_in * 1000
    };
    await privateJson(this.credentialPath(id), credentials);
    this.records = this.records.filter(row => row.id !== id).concat(record);
    await privateJson(path.join(this.directory, 'registrations.json'), this.records);
    this.active = { id, confirmed: false }; await privateJson(path.join(this.directory, 'active.json'), this.active);
    this.credentials = credentials;
    res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8', 'Cache-Control': 'no-store', 'Referrer-Policy': 'no-referrer' });
    res.end('<!doctype html><meta name="viewport" content="width=device-width"><title>登录完成</title><p>已连接你选择的 ChatGPT 账号。请回到回复助手，确认显示的是你自己的账号。</p>');
    this.finishPending();
  }

  async tokenRequest(fields) {
    const response = await this.fetch(`${ISSUER}/api/accounts/oauth/token`, { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: new URLSearchParams(fields), signal: AbortSignal.timeout(30000) });
    if (!response.ok) throw new Error('登录或续期失败，请重新登录自己的 ChatGPT 账号。');
    return response.json();
  }

  async accessToken() {
    if (!this.view().sharing || !this.view().confirmed) throw new Error('请先登录并确认这是你自己的 ChatGPT 账号，允许助手使用套餐额度。');
    if (this.expectedEmail && this.credentials.email?.toLowerCase() !== this.expectedEmail) throw new Error('当前登录的邮箱不是你指定的账号。');
    if (this.credentials.expires_at > Date.now() + 60000) return this.credentials.access_token;
    if (!this.refreshing) this.refreshing = this.refresh().finally(() => { this.refreshing = null; });
    await this.refreshing;
    return this.credentials.access_token;
  }

  async refresh() {
    const saved = this.credentials;
    if (!saved?.refresh_token) throw new Error('登录已到期，请重新登录。');
    const tokens = await this.tokenRequest({ grant_type: 'refresh_token', client_id: saved.client_id, refresh_token: saved.refresh_token, resource: RESOURCE });
    if (!tokens.access_token || !Number.isFinite(tokens.expires_in)) throw new Error('登录续期未完成，请重新登录。');
    const updated = { ...saved, access_token: tokens.access_token, refresh_token: tokens.refresh_token ?? saved.refresh_token, id_token: tokens.id_token ?? saved.id_token, scopes: tokens.scope ? tokens.scope.split(' ').filter(Boolean) : saved.scopes, expires_at: Date.now() + tokens.expires_in * 1000 };
    await privateJson(this.credentialPath(saved.id), updated);
    if (this.active?.id === saved.id) this.credentials = updated;
  }

  async signOut() {
    this.finishPending();
    if (this.refreshing) await this.refreshing.catch(() => {});
    const credentials = this.credentials;
    this.credentials = null;
    if (!credentials) return { remoteRevoked: true };
    let remoteRevoked = false;
    try {
      const discovery = await this.fetch(`${ISSUER}/.well-known/openid-configuration`, { signal: AbortSignal.timeout(10000) });
      const metadata = await discovery.json();
      const endpoint = new URL(metadata.revocation_endpoint);
      if (endpoint.origin !== ISSUER || endpoint.protocol !== 'https:') throw new Error('Invalid revocation endpoint');
      const response = await this.fetch(endpoint.href, { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: new URLSearchParams({ token: credentials.refresh_token, token_type_hint: 'refresh_token', client_id: credentials.client_id }), signal: AbortSignal.timeout(10000) });
      remoteRevoked = response.ok;
    } catch { /* Local sign-out still removes only this app's own credentials. */ }
    await fs.rm(this.credentialPath(credentials.id), { force: true });
    return { remoteRevoked };
  }
}

export function inferenceError(code) {
  if (code === 'subscription_sharing_usage_limit_exceeded') return '你的套餐额度暂时用完了，可以到 ChatGPT 查看重置时间。';
  if (code === 'subscription_sharing_usage_unavailable') return '这个账号目前未开放套餐接入，请到 ChatGPT 检查助手授权。';
  if (code === 'invalid_api_key' || code === 'invalid_token') return '登录已到期，请重新登录你自己的账号。';
  return '这次生成没有完成，请稍后再试。';
}

export async function consumeResponse(response) {
  if (!response.ok) {
    const error = await response.json().catch(() => ({}));
    throw new Error(inferenceError(error.error?.code));
  }
  const decoder = new TextDecoder();
  let buffer = '', text = '', completed = false, result = null;
  const parse = block => {
    const payload = block.split('\n').filter(line => line.startsWith('data:')).map(line => line.slice(5).trimStart()).join('\n');
    if (!payload || payload === '[DONE]') return;
    const event = JSON.parse(payload);
    if (event.type === 'response.output_text.delta') text += event.delta ?? '';
    if (event.type === 'response.failed' || event.type === 'error') throw new Error(inferenceError(event.response?.error?.code ?? event.code));
    if (event.type === 'response.incomplete') throw new Error('这次回复没有生成完整，请重试。');
    if (event.type === 'response.completed') { completed = true; result = event.response; }
  };
  for await (const chunk of response.body) {
    buffer += decoder.decode(chunk, { stream: true });
    buffer = buffer.replace(/\r\n/g, '\n');
    let boundary;
    while ((boundary = buffer.indexOf('\n\n')) >= 0) { parse(buffer.slice(0, boundary)); buffer = buffer.slice(boundary + 2); }
    if (buffer.length > 2_000_000 || text.length > 100_000) throw new Error('返回内容异常，请重新尝试。');
  }
  buffer += decoder.decode();
  if (buffer.trim()) parse(buffer);
  if (!completed) throw new Error('连接中断，这次生成未完成。');
  if (!text) text = (result?.output ?? []).flatMap(item => item.content ?? []).filter(part => part.type === 'output_text').map(part => part.text).join('');
  if (!text.trim()) throw new Error('没有收到完整回复，请重试。');
  return { text, usage: result?.usage ?? null };
}

export class ChatGPTInference {
  constructor(account, fetchImpl = fetch) { this.account = account; this.fetch = fetchImpl; }
  async models() {
    const token = await this.account.accessToken();
    const response = await this.fetch('https://api.openai.com/v1/models', { headers: { Authorization: `Bearer ${token}` }, signal: AbortSignal.timeout(30000) });
    if (!response.ok) throw new Error('还不能取得这个账号的可用模型，请检查登录和套餐授权。');
    const catalog = await response.json();
    if (!Array.isArray(catalog.models)) throw new Error('账号返回的模型列表格式不适用于套餐接入。');
    return catalog.models.filter(model => model.visibility === 'list').map(model => ({ id: model.slug, label: model.display_name ?? model.slug, inputModalities: model.input_modalities ?? null }));
  }
  async generate({ model, instructions, input, schema, signal }) {
    const token = await this.account.accessToken();
    const response = await this.fetch('https://api.openai.com/v1/responses', {
      method: 'POST', headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' },
      signal,
      body: JSON.stringify({ model, instructions, input, store: false, stream: true, ...(/^gpt-[56]/.test(model) ? { reasoning: { effort: 'low' } } : {}), text: { format: { type: 'json_schema', name: 'reply_assistant_output', strict: true, schema } } })
    });
    const result = await consumeResponse(response);
    return { ...result, value: JSON.parse(result.text) };
  }
}

export function validateImages(images = []) {
  if (!Array.isArray(images) || images.length > 3) throw new Error('一次最多放 3 张截图。');
  return images.map(image => {
    if (typeof image !== 'string' || image.length > 8_500_000) throw new Error('截图太大，请缩小后再试。');
    const match = image.match(/^data:(image\/(?:png|jpeg|webp));base64,([A-Za-z0-9+/]+={0,2})$/);
    if (!match) throw new Error('请使用 PNG、JPG 或 WebP 截图。');
    const bytes = Buffer.from(match[2], 'base64');
    const png = bytes.subarray(0, 8).equals(Buffer.from([137,80,78,71,13,10,26,10]));
    const jpeg = bytes[0] === 255 && bytes[1] === 216 && bytes[2] === 255;
    const webp = bytes.subarray(0,4).toString() === 'RIFF' && bytes.subarray(8,12).toString() === 'WEBP';
    if (!({ 'image/png': png, 'image/jpeg': jpeg, 'image/webp': webp })[match[1]]) throw new Error('截图格式与内容不一致。');
    return { type: 'input_image', image_url: image };
  });
}

import fs from 'node:fs/promises';
import path from 'node:path';
import zlib from 'node:zlib';
import crypto from 'node:crypto';

export const normalize = text => text.normalize('NFKC').replace(/\s+/g, '').replaceAll('刘子睿', 'ta');
export const alias = value => ['刘子睿', 'Rui'].includes(value) ? 'ta' : value === 'Yee' ? '我' : value;
export const cleanName = text => text.replaceAll('刘子睿', 'ta');

function decode(text) {
  return text.replace(/&(#x[\da-fA-F]+|#\d+|amp|lt|gt|quot|apos);/g, (match, entity) => {
    const known = { amp: '&', lt: '<', gt: '>', quot: '"', apos: "'" };
    return known[entity] ?? String.fromCodePoint(entity[1] === 'x' ? parseInt(entity.slice(2), 16) : parseInt(entity.slice(1), 10));
  });
}

export function unzip(buffer) {
  let end = buffer.length - 22;
  const minimum = Math.max(0, buffer.length - 65557);
  while (end >= minimum && buffer.readUInt32LE(end) !== 0x06054b50) end--;
  if (end < minimum) throw new Error('Invalid DOCX ZIP');
  const count = buffer.readUInt16LE(end + 10);
  let offset = buffer.readUInt32LE(end + 16);
  const entries = new Map();
  for (let i = 0; i < count; i++) {
    if (buffer.readUInt32LE(offset) !== 0x02014b50) throw new Error('Invalid ZIP central directory');
    const method = buffer.readUInt16LE(offset + 10);
    const size = buffer.readUInt32LE(offset + 20);
    const nameLength = buffer.readUInt16LE(offset + 28);
    const extraLength = buffer.readUInt16LE(offset + 30);
    const commentLength = buffer.readUInt16LE(offset + 32);
    const localOffset = buffer.readUInt32LE(offset + 42);
    const name = buffer.subarray(offset + 46, offset + 46 + nameLength).toString('utf8');
    const start = localOffset + 30 + buffer.readUInt16LE(localOffset + 26) + buffer.readUInt16LE(localOffset + 28);
    const packed = buffer.subarray(start, start + size);
    if (![0, 8].includes(method)) throw new Error('Unsupported DOCX compression');
    entries.set(name, method === 8 ? zlib.inflateRawSync(packed) : Buffer.from(packed));
    offset += 46 + nameLength + extraLength + commentLength;
  }
  return entries;
}

export function paragraphs(buffer) {
  const document = unzip(buffer).get('word/document.xml');
  if (!document) throw new Error('DOCX document.xml missing');
  return [...document.toString().matchAll(/<w:p\b[^>]*>[\s\S]*?<\/w:p>/g)].map((match, i) => {
    const xml = match[0];
    const runs = [...xml.matchAll(/<w:r\b[^>]*>[\s\S]*?<\/w:r>/g)].map(run => ({
      text: [...run[0].matchAll(/<w:t(?:\s[^>]*)?>([\s\S]*?)<\/w:t>/g)].map(t => decode(t[1])).join(''),
      tab: /<w:tab\b/.test(run[0]),
      highlighted: /<w:highlight\b[^>]*w:val="yellow"/.test(run[0])
    }));
    return {
      paragraph: i + 1,
      style: xml.match(/<w:pStyle\b[^>]*w:val="([^"]*)"/)?.[1] ?? '',
      text: runs.map(run => run.text + (run.tab ? '\t' : '')).join(''),
      runs
    };
  });
}

export function selectedReplies(editedParagraphs, rows) {
  const lookup = new Map();
  for (const row of rows) {
    const key = normalize(row.content);
    lookup.set(key, [...(lookup.get(key) ?? []), row]);
  }
  const selected = new Map();
  const unresolved = [];
  let appearances = 0;
  for (const paragraph of editedParagraphs) {
    const text = paragraph.text.trim();
    if (!text || /^引用\s*(我|ta)[：:]/.test(text) || paragraph.style === '2') continue;
    const labelled = text.match(/^(\d{2}:\d{2}:\d{2})\t(我|ta|系统提示)\t([\s\S]*)$/);
    const body = labelled?.[3] ?? text;
    let candidates = lookup.get(normalize(body)) ?? [];
    if (labelled) candidates = candidates.filter(row => row.time === labelled[1] && alias(row.sender) === labelled[2]);
    const owners = [...new Set(candidates.map(row => alias(row.sender)))];
    const own = labelled?.[2] === '我' || (!labelled && owners.length === 1 && owners[0] === '我');
    const highlighted = paragraph.runs.filter(run => run.highlighted).map(run => run.text).join('');
    if (!own && !(highlighted && normalize(highlighted) === normalize(body))) continue;
    candidates = candidates.filter(row => row.sender === '我');
    if (candidates.length !== 1) {
      unresolved.push({ paragraph: paragraph.paragraph, candidate_count: candidates.length });
      continue;
    }
    appearances++;
    const row = candidates[0];
    selected.set(row.index, [...(selected.get(row.index) ?? []), paragraph.paragraph]);
  }
  return { selected, appearances, unresolved };
}

export function buildTurns(rows, gapSeconds = 120, excluded = () => false) {
  const turns = [];
  let current = null;
  for (const row of rows) {
    if (excluded(row)) { current = null; continue; }
    if (row.sender === '系统提示') continue;
    const previous = current;
    if (previous && previous.speaker === alias(row.sender) && row.timestamp - previous.last_timestamp <= gapSeconds) {
      previous.records.push(row);
      previous.last_timestamp = row.timestamp;
    } else {
      current = { speaker: alias(row.sender), records: [row], last_timestamp: row.timestamp };
      turns.push(current);
    }
  }
  return turns;
}

function contextText(text) {
  return cleanName(text)
    .replace(/https?:\/\/\S+/g, '[链接]')
    .replace(/\b\d{17}[\dXx]\b/g, '[身份号码]')
    .replace(/\b1[3-9]\d{9}\b/g, '[电话号码]')
    .replace(/[^\n]*?(?:街道|号楼|单元|家园|华府)[^\n]*/g, '[详细地址]');
}

function publicRecord(row) {
  return {
    source_index: row.index,
    date: row.date,
    time: row.time,
    speaker: alias(row.sender),
    text: contextText(row.content),
    quote: row.quote ? { speaker: alias(row.quote.sender), text: contextText(row.quote.content) } : null
  };
}

function reuseChecks(text) {
  const checks = [];
  if (/\d{5,}|https?:\/\/|街道|号楼|单元|账号|地址/.test(text)) checks.push('identifier_or_address');
  if (/刘总|小刘|京爷|北京人|女明星|哥哥|妹妹|年上|年下|crush|前夫/.test(text)) checks.push('relationship_specific_address');
  if (/北京|香港|深圳|广州|长沙|北海|纽约|江门|宝安|南山|福田/.test(text)) checks.push('place_or_identity');
  if (/礼物|送你|送我|牵手|上手|梦到|我朋友|我闺蜜|我爸|我妈|明天|明晚|下周|周五|周六|周日|昨天|生日|我家/.test(text)) checks.push('current_facts_required');
  return checks;
}

export async function buildCorpus({ originalDocx, editedDocx, recordsJson, outputDir, exclusions = { source_index_ranges: [] } }) {
  const [originalBytes, editedBytes, recordsBytes] = await Promise.all([
    fs.readFile(originalDocx), fs.readFile(editedDocx), fs.readFile(recordsJson)
  ]);
  const rows = JSON.parse(recordsBytes).records;
  const originalParagraphs = paragraphs(originalBytes).filter(p => p.style === 'ChatMessage');
  if (originalParagraphs.length !== rows.length) throw new Error('Original Word message count mismatch');
  for (let i = 0; i < rows.length; i++) {
    if (originalParagraphs[i].text.replaceAll('\t', '') !== rows[i].time + rows[i].sender + rows[i].content) {
      throw new Error(`Original Word content mismatch at row ${i}`);
    }
  }
  const preference = selectedReplies(paragraphs(editedBytes), rows);
  if (preference.unresolved.length) throw new Error(`Unresolved preferred reply attribution: ${JSON.stringify(preference.unresolved)}`);
  const isExcluded = row => exclusions.source_index_ranges.some(range => row.index >= range.start && row.index <= range.end);
  const excludedQuoteTexts = new Set(rows.filter(isExcluded).map(row => normalize(row.content)));
  const toContext = row => {
    const record = publicRecord(row);
    if (row.quote && excludedQuoteTexts.has(normalize(row.quote.content))) record.quote = null;
    return record;
  };
  const originalOwnTurns = buildTurns(rows).filter(turn => turn.speaker === '我');
  const turns = buildTurns(rows, 120, isExcluded);
  const ownTurns = turns.filter(t => t.speaker === '我');
  const samples = [];
  const positions = new Map(rows.map((row, i) => [row.index, i]));
  for (const [number, turn] of ownTurns.entries()) {
    const before = positions.get(turn.records[0].index);
    const after = positions.get(turn.records.at(-1).index);
    const first = turn.records[0];
    const sourceIds = turn.records.map(row => row.index);
    const checks = [...new Set(turn.records.flatMap(row => reuseChecks(row.content)))];
    samples.push({
      id: `turn_${String(first.index).padStart(4, '0')}`,
      date: first.date,
      time: first.time,
      reply: turn.records.map(row => cleanName(row.content)),
      source_indices: sourceIds,
      preferred_source_indices: sourceIds.filter(index => preference.selected.has(index)),
      preferred_paragraphs: sourceIds.flatMap(index => preference.selected.get(index) ?? []),
      context_before: rows.slice(Math.max(0, before - 6), before).filter(row => row.sender !== '系统提示' && !isExcluded(row)).map(toContext),
      addressed_messages: turn.records.filter(row => row.quote && !excludedQuoteTexts.has(normalize(row.quote.content))).map(row => ({ replying_source_index: row.index, ...toContext(row).quote })),
      continuation_after: rows.slice(after + 1, after + 5).filter(row => row.sender !== '系统提示' && !isExcluded(row)).map(toContext),
      reuse_checks: checks,
      direct_reuse_blocked: checks.includes('identifier_or_address'),
      original_relationship_stage: 'not_assigned'
    });
  }
  const lengths = rows.filter(row => row.sender === '我').map(row => [...row.content].length).sort((a, b) => a - b);
  const stats = {
    source_record_count: rows.length,
    own_message_count: lengths.length,
    other_message_count: rows.filter(row => row.sender === '刘子睿').length,
    system_message_count: rows.filter(row => row.sender === '系统提示').length,
    own_text_turn_count: samples.length,
    original_own_text_turn_count: originalOwnTurns.length,
    eligible_own_message_count: rows.filter(row => row.sender === '我' && !isExcluded(row)).length,
    excluded_own_message_count: rows.filter(row => row.sender === '我' && isExcluded(row)).length,
    excluded_source_ranges: exclusions.source_index_ranges,
    grouping_rule: 'Consecutive text messages from the same speaker, separated by at most 120 seconds; system notices ignored.',
    preferred_appearances: preference.appearances,
    preferred_unique_messages: preference.selected.size,
    eligible_preferred_unique_messages: [...preference.selected.keys()].filter(index => !isExcluded({ index })).length,
    duplicate_preference_appearances: preference.appearances - preference.selected.size,
    own_char_median: lengths[Math.floor(lengths.length / 2)],
    own_messages_at_most_20_chars: lengths.filter(n => n <= 20).length,
    first_date: rows[0].date,
    last_date: rows.at(-1).date,
    source_verified_against_docx: true,
    sources: [originalDocx, editedDocx].map((filename, i) => ({ filename: path.basename(filename), sha256: crypto.createHash('sha256').update(i ? editedBytes : originalBytes).digest('hex') })),
    unresolved_preferences: preference.unresolved,
    conversation_context_scrubbed: ['telephone_numbers', 'identity_numbers', 'detailed_addresses', 'URL_values']
  };
  await fs.mkdir(outputDir, { recursive: true, mode: 0o700 });
  const save = (name, body) => fs.writeFile(path.join(outputDir, name), body, { mode: 0o600 });
  await save('reply_turns.json', JSON.stringify(samples, null, 2));
  await save('conversation.json', JSON.stringify(rows.filter(row => !isExcluded(row)).map(toContext), null, 2));
  await save('stats.json', JSON.stringify(stats, null, 2));
  return { stats, samples };
}

import fs from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { normalize } from './corpus.mjs';

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const stageAliases = { '刚加好友': 'new', '加好友': 'new', '认识一段时间': 'familiar', '加了一段时间后': 'familiar', '即将在一起': 'close' };
const stageNames = { new: '刚加好友', familiar: '认识一段时间', close: '即将在一起' };

function grams(text) {
  const words = normalize(text).toLowerCase().match(/[\p{Script=Han}a-z0-9]+/gu) ?? [];
  const tokens = new Set();
  for (const word of words) {
    const chars = [...word];
    if (chars.length === 1) tokens.add(word);
    for (let i = 0; i + 1 < chars.length; i++) tokens.add(chars[i] + chars[i + 1]);
  }
  return tokens;
}

function overlap(query, document, weights) {
  let numerator = 0, queryWeight = 0, documentWeight = 0;
  for (const token of query) queryWeight += weights.get(token) ?? 1;
  for (const token of document) documentWeight += weights.get(token) ?? 1;
  for (const token of query) if (document.has(token)) numerator += weights.get(token) ?? 1;
  return numerator / Math.max(1, Math.sqrt(queryWeight * documentWeight));
}

export function forbiddenThirdPartyClaim(text) {
  return /准备[^。\n]{0,25}(?:部门|同事|留子)[^。\n]{0,20}(?:对象|男朋友)|帅哥[^。\n]{0,15}(?:帮我|封网)|(?:找|介绍|认识)[^。\n]{0,10}帅哥|哥哥们|几个哥哥|全是年上/.test(text);
}

function inferredMinimumStage(text) {
  if (/牵手|地下恋|占有欲|关键情节|一心只有|惦记奖励/.test(text)) return 'close';
  if (/报备|crush|想我|想你|哥哥|妹妹|仅你可见|奖励/.test(text)) return 'familiar';
  return 'new';
}

function invitationState(query, availability, suppliedState) {
  const valid = ['need_invitation', 'need_date', 'need_time', 'offer_availability', 'confirm'];
  if (suppliedState) {
    if (!valid.includes(suppliedState)) throw new Error('Unknown invitation state');
    return suppliedState;
  }
  if (/还(?:没有|没)(?:主动)?(?:提出|提过|说过)?(?:邀约|邀请|约我|要带我)|尚未(?:邀约|邀请)|还没说(?:要)?带我/.test(query)) return 'need_invitation';
  if (/(还没|还没有|未|没定|没有确定)[^。\n]{0,15}(几点|钟点)/.test(query)) return 'need_time';
  if (availability.length && /哪天|有空|时间窗口|选|挑/.test(query)) return 'offer_availability';
  if (/改天|下次有空|具体哪天|定具体时间|确定哪天/.test(query)) return 'need_date';
  return 'need_invitation';
}

export function retrieveCandidates(library, { query, stage = 'familiar', emotion = 'unknown', limit = 5, includeUnreviewed = true, goal = null, availability = [], bookingState = null }) {
  stage = stageAliases[stage] ?? stage;
  if (!stageNames[stage]) throw new Error('Stage must be new, familiar, close, or its Chinese label');
  if (!['unknown', 'playful', 'warm', 'unhappy'].includes(emotion)) throw new Error('Unknown emotion mode');
  if (!query?.trim()) throw new Error('A nonempty conversation query is required');
  if (goal && !library.goals[goal]) throw new Error('Unknown reply goal');
  const currentBookingState = goal === 'concrete_invitation' ? invitationState(query, availability, bookingState) : null;
  const bookingSteps = {
    need_invitation: ['opening', 'ordinary_topic_bridge'],
    need_date: ['request_date'],
    need_time: ['request_time'],
    offer_availability: ['offer_real_availability', 'let_him_choose'],
    confirm: ['confirm_date', 'confirm_time']
  };
  if (emotion === 'unknown' && /生气|不高兴|争执|吵架|不真诚|不信任|不回答只提问|别再这样|玩笑还是/.test(query)) {
    return { status: 'clarify_emotion', stage: stageNames[stage], emotion, goal: goal ? { key: goal, ...library.goals[goal] } : null, booking_state: currentBookingState, user_provided_availability: availability, clarification: '这里是在互相开玩笑，还是有人真的不高兴？确认后再选择斗嘴或温和回应。', candidates: [] };
  }
  const excluded = index => library.exclusions.source_index_ranges.some(range => index >= range.start && index <= range.end);
  const pool = library.cases.map(example => ({ ...example, annotated: true }));
  if (includeUnreviewed) {
    const ranks = { new: 0, familiar: 1, close: 2 };
    for (const turn of library.turns) {
      if (turn.direct_reuse_blocked || turn.source_indices.some(excluded) || turn.reply.some(forbiddenThirdPartyClaim)) continue;
      if (ranks[inferredMinimumStage(turn.reply.join(' '))] > ranks[stage]) continue;
      pool.push({ ...turn, preferred: turn.preferred_source_indices.length > 0, annotated: false, scene: '', retrieval_hints: [], review_status: 'needs_current_scene_review', mode: 'unreviewed' });
    }
  }
  const valid = pool.filter(candidate => {
    if (candidate.source_indices.some(excluded) || candidate.reply.some(forbiddenThirdPartyClaim)) return false;
    if (goal && !candidate.supported_goals?.includes(goal)) return false;
    if (currentBookingState === 'need_invitation' && !candidate.booking_step) return false;
    if (currentBookingState && candidate.booking_step && !bookingSteps[currentBookingState].includes(candidate.booking_step)) return false;
    if (currentBookingState === 'need_date' && !candidate.booking_step) return false;
    if (currentBookingState === 'need_time' && !candidate.booking_step) return false;
    if (candidate.allowed_stages && !candidate.allowed_stages.includes(stage)) return false;
    if (['warm', 'unhappy'].includes(emotion) && candidate.mode !== 'warm') return false;
    return true;
  });
  const inputs = valid.map(candidate => {
    const before = [...(candidate.context_before ?? []), ...(candidate.addressed_messages ?? [])].map(row => row.text).join(' ');
    return { candidate, context: grams(before), intent: grams([candidate.scene, ...(candidate.retrieval_hints ?? [])].join(' ')), reply: grams(candidate.reply.join(' ')) };
  });
  const frequencies = new Map();
  for (const input of inputs) {
    for (const token of new Set([...input.context, ...input.intent])) frequencies.set(token, (frequencies.get(token) ?? 0) + 1);
  }
  const weights = new Map([...frequencies].map(([token, frequency]) => [token, 1 + Math.log((inputs.length + 1) / (frequency + 1))]));
  const q = grams(query);
  const normalizedQuery = normalize(query);
  const scored = inputs.map(input => {
    const hints = input.candidate.retrieval_hints.filter(hint => normalizedQuery.includes(normalize(hint)));
    const hintScore = Math.min(4, hints.reduce((sum, hint) => sum + 0.5 + Math.min(1.5, [...hint].length / 4), 0));
    const contextScore = overlap(q, input.context, weights);
    const intentScore = overlap(q, input.intent, weights);
    const replyScore = overlap(q, input.reply, weights);
    const score = (contextScore * 2 + intentScore * 1.2 + replyScore * 0.12 + hintScore) * (input.candidate.preferred ? 1.06 : 1);
    return { ...input.candidate, score, matched_hints: hints, context_score: contextScore };
  }).filter(candidate => candidate.annotated
    ? candidate.score >= 0.45 && (candidate.matched_hints.length > 0 || candidate.context_score >= 0.18)
    : candidate.score >= 0.55 && candidate.context_score >= 0.2
  ).sort((a, b) => b.score - a.score || Number(b.annotated) - Number(a.annotated));
  const accepted = [];
  for (const candidate of scored) {
    if (candidate.score < Math.max(0.55, (scored[0]?.score ?? 0) * 0.35)) continue;
    if (accepted.some(chosen => candidate.source_indices.some(index => chosen.source_indices.includes(index)))) continue;
    accepted.push(candidate);
    if (accepted.length >= Math.max(1, Math.min(20, limit))) break;
  }
  return {
    status: accepted.length ? 'candidates_for_context_review' : 'no_close_match',
    stage: stageNames[stage],
    emotion,
    goal: goal ? { key: goal, ...library.goals[goal] } : null,
    booking_state: currentBookingState,
    user_provided_availability: availability,
    ranking_method: 'Local Chinese character-bigram and curated scene-hint retrieval. Scores do not establish suitability.',
    candidates: accepted.map(candidate => ({
      id: candidate.id, source_indices: candidate.source_indices, source_type: 'original', date: candidate.date,
      reply: candidate.reply, preferred: candidate.preferred, scene: candidate.scene || null,
      condition: candidate.condition || '未逐场景标注，先结合当前上下文确认适合程度。',
      inferred_subtext: candidate.inferred_subtext || null,
      reviewed: candidate.annotated, score: Number(candidate.score.toFixed(4)), matched_hints: candidate.matched_hints,
      booking_step: candidate.booking_step ?? null,
      reuse_checks: candidate.reuse_checks ?? [],
      context_before: candidate.context_before,
      addressed_messages: candidate.addressed_messages,
      continuation_after: candidate.continuation_after
    }))
  };
}
